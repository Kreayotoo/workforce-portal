CREATE TABLE IF NOT EXISTS work_logs ( id uuid DEFAULT gen_random_uuid() PRIMARY KEY, created_at timestamptz DEFAULT now(), updated_at timestamptz, emp_uuid uuid, emp_name text, emp_email text, log_date date, mode text, entries jsonb DEFAULT '[]'::jsonb, summary text, tags jsonb DEFAULT '[]'::jsonb, admin_comments jsonb DEFAULT '[]'::jsonb, total_hours numeric, project_entries jsonb DEFAULT '[]'::jsonb, project_total_hours numeric, company_id text ); ALTER TABLE work_logs DISABLE ROW LEVEL SECURITY;
